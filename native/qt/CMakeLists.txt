cmake_minimum_required(VERSION 3.24)
project(GoLiveQt VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(GOLIVE_STATIC_ANALYSIS "Run MSVC code analysis during compilation" OFF)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network WebSockets Test)
add_library(golive-core STATIC protocol.cpp engine.cpp room.cpp)
target_link_libraries(golive-core PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::WebSockets)
target_include_directories(golive-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(golive-core PUBLIC GOLIVE_VERSION="0.6.0-qt.4" NOMINMAX WIN32_LEAN_AND_MEAN)
add_executable(golive-qt WIN32 main.cpp window.cpp video.cpp player.cpp icons.cpp updater.cpp thumbnails.cpp resources.qrc)
target_link_libraries(golive-qt PRIVATE golive-core Qt6::Widgets)
if(WIN32)
  target_sources(golive-qt PRIVATE app-icon.rc)
  target_link_libraries(golive-qt PRIVATE dwmapi)
endif()
set_target_properties(golive-qt PROPERTIES OUTPUT_NAME "GoLive P2P")
if(MSVC)
  target_compile_options(golive-core PRIVATE /W4 /WX /utf-8)
  target_compile_options(golive-qt PRIVATE /W4 /WX /utf-8)
  if(GOLIVE_STATIC_ANALYSIS)
    target_compile_options(golive-core PRIVATE /analyze)
    # Qt forward-declares HRESULT before Windows SDK SAL declarations; this
    # annotation mismatch is external to the application.
    target_compile_options(golive-qt PRIVATE /analyze /wd28301)
  endif()
endif()
enable_testing()
if(UNIX AND NOT APPLE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-1.0 gstreamer-app-1.0 gstreamer-webrtc-1.0 gstreamer-sdp-1.0 gstreamer-video-1.0)
  add_executable(media-engine ../windows/media-engine.cpp)
  target_link_libraries(media-engine PRIVATE PkgConfig::GST)
  target_compile_options(media-engine PRIVATE -Wall -Wextra)
endif()
add_executable(qt-tests tests.cpp)
target_link_libraries(qt-tests PRIVATE golive-core Qt6::Test)
add_test(NAME qt-core COMMAND qt-tests -o qt-tests.txt,txt)
add_executable(player-tests player-tests.cpp player.cpp video.cpp icons.cpp)
target_link_libraries(player-tests PRIVATE golive-core Qt6::Widgets Qt6::Test)
add_test(NAME qt-player COMMAND player-tests -o player-tests.txt,txt)
set_tests_properties(qt-player PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
if(UNIX AND NOT APPLE)
  add_executable(receive-buffer-test ../windows/receive-buffer-test.cpp)
  target_link_libraries(receive-buffer-test PRIVATE PkgConfig::GST)
  add_test(NAME rtp-receive-buffer COMMAND receive-buffer-test)
endif()

add_executable(audio-continuity-test ../windows/audio-continuity-test.cpp)
add_test(NAME audio-continuity COMMAND audio-continuity-test)

add_executable(engine-fixture engine-fixture.cpp)
target_link_libraries(engine-fixture PRIVATE Qt6::Core)
add_dependencies(qt-tests engine-fixture)
if(UNIX AND NOT APPLE)
  add_executable(opus-continuity-test ../windows/opus-continuity-test.cpp)
  target_link_libraries(opus-continuity-test PRIVATE PkgConfig::GST)
  add_test(NAME opus-continuity COMMAND opus-continuity-test)
  add_executable(srtp-runtime-test ../windows/srtp-runtime-test.cpp)
  target_link_libraries(srtp-runtime-test PRIVATE PkgConfig::GST)
  add_test(NAME srtp-runtime COMMAND srtp-runtime-test)
endif()
